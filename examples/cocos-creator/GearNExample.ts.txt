/**
 * GearNExample.ts - Example Cocos Creator 3.x Component using GearN SDK
 *
 * This component demonstrates how to integrate the GearN SDK
 * with a Cocos Creator 3.x project.
 *
 * Installation:
 * 1. npm install @xmobitea/gn-typescript-client
 * 2. Copy this file to your Cocos Creator project's scripts folder
 * 3. Attach to a Node in your scene
 */

import { _decorator, Component, error, log, warn } from "cc";
import {
    AuthenticateModels,
    ErrorCode,
    GNNetwork,
    GNServerSettings,
    LogType,
    MessageType,
} from "@xmobitea/gn-typescript-client";

const { ccclass, property } = _decorator;

@ccclass("GearNExample")
export class GearNExample extends Component {
    @property({ tooltip: "GearN Server host or IP" })
    serverAddress: string = "127.0.0.1";

    @property({ tooltip: "GearN HTTP port" })
    serverPort: number = 2202;

    @property({ tooltip: "GearN socket port" })
    serverSocketPort: number = 2901;

    @property({ tooltip: "Client secret key" })
    secretKey: string = "";

    @property({ tooltip: "Game id" })
    gameId: string = "";

    @property({ tooltip: "Use HTTPS public endpoint" })
    useSsl: boolean = false;

    @property({ tooltip: "Enable debug logging" })
    enableDebugLog: boolean = true;

    start(): void {
        this.initializeGearN();
    }

    /**
     * Initialize GearN SDK with the current component settings.
     */
    private initializeGearN(): void {
        try {
            const settings = new GNServerSettings();
            settings.config({
                serverAddress: this.serverAddress,
                serverPort: this.serverPort,
                serverSocketPort: this.serverSocketPort,
                useSsl: this.useSsl,
                useSocket: true,
                useHttp: true,
                defaultTimeoutInSeconds: 20,
                messageType: MessageType.MsgPack,
                sendRate: 30,
                reconnectDelay: 1000,
                pingInterval: 3000,
                pingTimeout: 5000,
                secretKey: this.secretKey,
                gameId: this.gameId,
                logType: this.enableDebugLog ? LogType.All : LogType.None,
            });

            GNNetwork.init(settings);

            log("[GearN] SDK initialized successfully!");
            log("[GearN] Platform:", GNNetwork.getPlatform());

            this.setupConnectionHandlers();
        } catch (err) {
            error("[GearN] Failed to initialize SDK:", err);
        }
    }

    /**
     * Register socket lifecycle callbacks for debugging and UI hooks.
     */
    private setupConnectionHandlers(): void {
        GNNetwork.subscriberOnConnectHandler(() => {
            log("[GearN] Socket connected! SID:", GNNetwork.getSocketSId());
        });

        GNNetwork.subscriberOnDisconnectHandler(() => {
            warn("[GearN] Socket disconnected!");
        });
    }

    /**
     * Connect to the socket server.
     * The SDK auto-authenticates the socket if an auth token is already cached.
     */
    public connectSocket(): void {
        log("[GearN] Connecting to socket...");
        GNNetwork.connectSocket();
    }

    /**
     * Disconnect from the socket server.
     */
    public disconnectSocket(): void {
        log("[GearN] Disconnecting from socket...");
        GNNetwork.disconnectSocket();
    }

    /**
     * Login with a custom device id and then connect the socket.
     */
    public async loginWithCustomDeviceId(deviceId: string): Promise<void> {
        try {
            log("[GearN] Logging in with custom device ID:", deviceId);

            const infoRequestParam = new AuthenticateModels.InfoRequestParam();
            infoRequestParam.displayName = true;
            infoRequestParam.avatar = true;

            const request = new AuthenticateModels.LoginByCustomDeviceIdRequestData();
            request.customDeviceId = deviceId;
            request.createPlayerIfNotExists = true;
            request.infoRequestParam = infoRequestParam;

            const response = await GNNetwork.authenticate.loginByCustomDeviceIdAsync(request);

            if (response.hasReturnCodeError()) {
                throw new Error(response.debugMessage);
            }

            if (response.errorCode !== ErrorCode.Ok) {
                throw new Error(`Business error: ${response.errorCode}`);
            }

            log("[GearN] Login successful!");
            log("[GearN] User ID:", GNNetwork.getAuthenticateStatus().getUserId());

            this.connectSocket();
        } catch (err) {
            error("[GearN] Login error:", err);
        }
    }

    /**
     * Re-auth the current socket when it was connected before the token existed,
     * or after the token has changed and reconnect is not desired.
     */
    public reAuthSocketIfNeeded(): void {
        if (!GNNetwork.isSocketConnected()) {
            warn("[GearN] Socket is not connected");
            return;
        }

        if (!GNNetwork.getAuthenticateStatus().getAuthToken()) {
            warn("[GearN] No auth token is currently cached");
            return;
        }

        GNNetwork.sendRequestAuthSocket();
        log("[GearN] Socket re-auth request sent");
    }

    /**
     * Example: Check whether the server is reachable.
     */
    public async checkServerHealth(): Promise<void> {
        try {
            const response = await GNNetwork.healthCheckAsync();
            log("[GearN] Server health:", response);
        } catch (err) {
            error("[GearN] Health check failed:", err);
        }
    }

    onDestroy(): void {
        if (GNNetwork.isSocketConnected()) {
            GNNetwork.disconnectSocket();
        }
    }
}
